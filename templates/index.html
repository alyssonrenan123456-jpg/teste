async function abrirResumoPlantao() { const modal = document.getElementById('modal-resumo'); const conteudo = document.getElementById('modal-conteudo'); modal.classList.remove('hidden'); conteudo.innerHTML = `

Carregando todas as cidades e escalas...
`; try { const response = await fetch('/api/buscar', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ termo: 'TODAS_AS_CIDADES', tipo: 'plantao' }) }); const data = await response.json(); if (!data.sucesso || data.dados.length === 0) { conteudo.innerHTML = `
Nenhum dado encontrado para gerar o resumo.
`; return; } let tabelaHtml = `
`; data.dados.forEach(item => { const temSab = item.tecnico_sabado && item.tecnico_sabado !== 'Nenhum técnico escalado' && item.tecnico_sabado !== 'NENHUMA OPÇÃO'; const temDom = item.tecnico_domingo && item.tecnico_domingo !== 'Nenhum técnico escalado' && item.tecnico_domingo !== 'NENHUMA OPÇÃO'; tabelaHtml += ` `; }); tabelaHtml += `
Filial / Cidade Sábado Domingo
${item.cidade} ${item.filial} ${temSab ? 'Sim' : 'Não'} ${temDom ? 'Sim' : 'Não'}
`; conteudo.innerHTML = tabelaHtml; } catch (e) { conteudo.innerHTML = `
Erro ao carregar o resumo da escala.
`; } }
